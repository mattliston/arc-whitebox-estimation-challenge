

# BEGIN OPUS-GOVERNOR FRAGMENT slow_governor_v1 (2026-10-16; append LAST on S2tr 46e9315d)
# Residual governor. The platform fails a network whose residual (the participant's Python outside flopscope calls) passes 0.4 s.
# At the start of layers _GOV_LAYERS this reads the network's residual so far from the active flopscope BudgetContext's live
# residual_wall_time_s (a local clock read, no server request) and projects the end residual as residual / _GOV_FRAC[layer], the
# nominal share spent before that layer (boundary traces). Above _GOV_CAP_S the rest of the network runs request-cheap:
# 'priced' = the tree planners get their pre-prices_zero prices (their plans cached apart from the zero-price plans);
# 'full' / 'two' = each eligible covariance sandwich is one full product / the SAFE file's two strips.
# Same statistical model, rounding-level change. A network that does not switch runs exactly as S2tr (same requests and bytes).
_GOV_CAP_S = 0.385
_GOV_LAYERS = (4, 14)
_GOV_FRAC = (0.0, 0.0625, 0.125, 0.1875, 0.25, 0.3125, 0.375, 0.4375, 0.5, 0.5625, 0.625, 0.6875, 0.75, 0.8125, 0.875, 0.9375)
_GOV_ACTIONS = ('priced', 'full')
_GOV_FORCE = None            # tests only: the layer start that switches whatever the projection
_GOV_PRICES = dict(_DR_MU=4e8, _DR_MU_SQ=6e8, _READ6_MU=3.5e8, _P6_MU=3e8, _KS_PRICE=7.5e5, _HZ_PRICE=2.5e5, _HZG_PRICE=7.5e5)
_GOV_CACHES = ('_DR_CACHE', '_READ6_CHOICE', '_P6_PLANS', '_KS_PLANS', '_HZ_PLANS', '_HZG_PLANS')
assert all(globals()[name] == 0.0 for name in _GOV_PRICES)
_GOV_ZERO = {name: globals()[name] for name in tuple(_GOV_PRICES) + _GOV_CACHES}
_GOV_PRICED = dict(_GOV_PRICES, **{name: {} for name in _GOV_CACHES})
_GOV = dict(on=False, r0=None, layer=None, projected=None)
_GOV_PARENT_SYM = _sym_upper
_GOV_PARENT_PSDM = _psdm_sym_upper
_GOV_PARENT_BEGIN = _w5_port_begin
_GOV_PARENT_PREDICT = Estimator.predict


def _gov_residual():
    try:
        r = flops._budget._active_context.residual_wall_time_s
    except Exception:
        return None
    return r if type(r) is float else None


def _gov_switch(on):
    globals().update(_GOV_PRICED if on and 'priced' in _GOV_ACTIONS else _GOV_ZERO)
    _GOV['on'] = on


def _w5_port_begin(scope, network, layer, weight, previous):
    if not _GOV['on']:
        if _GOV_FORCE is not None:
            if layer == _GOV_FORCE:
                _GOV.update(layer=layer)
                _gov_switch(True)
        elif _GOV['r0'] is not None and _GOV_LAYERS[0] <= layer <= _GOV_LAYERS[1]:
            r = _gov_residual()
            if r is not None and r - _GOV['r0'] > _GOV_CAP_S * _GOV_FRAC[layer]:
                _GOV.update(layer=layer, projected=(r - _GOV['r0']) / _GOV_FRAC[layer])
                _gov_switch(True)
    return _GOV_PARENT_BEGIN(scope, network, layer, weight, previous)


def _gov_fits(A, B, inner, rows):
    return (_GOV['on'] and _AB_ON and _WINOGRAD and _SCRATCH is not None
            and inner in (896, 960, 1024)
            and len(A.shape) == len(B.shape) == 2
            and tuple(A.shape) == (1024, inner)
            and tuple(B.shape) == (inner, 1024)
            and str(A.dtype) == str(B.dtype) == 'float32'
            and all(_ipt_fits(6, m, inner, n) for m, n in rows))


def _gov_full(A, B):
    # One 1024-square product; G10 mirrored from G01 as the parent's upper-block semantics.
    keep = []
    _TR_LAST[0] = None
    try:
        out = fnp.empty((1024, 1024), dtype=fnp.float32)
        _ipt_product_ab(A, B, out, 6)
        upper, lower = out[:512, 512:], out[512:, :512]
        upper_t = upper.T
        keep.extend((upper, lower, upper_t))
        keep.append(fnp.copyto(lower, upper_t))
        return out
    finally:
        _TR_LAST[0] = None
        keep.clear()


def _gov_two(A, B):
    # The SAFE file's two strips: [G00 G01] and G11, G10 = G01.T.
    keep = []
    _TR_LAST[0] = None
    try:
        out = fnp.empty((1024, 1024), dtype=fnp.float32)
        a0, a1, b1 = A[:512, :], A[512:, :], B[:, 512:]
        top = fnp.empty((512, 1024), dtype=fnp.float32)
        bottom = fnp.empty((512, 512), dtype=fnp.float32)
        keep.extend((a0, a1, b1, top, bottom))
        _ipt_product_ab(a0, B, top, 6)
        upper, lower, cross = out[:512, :], out[512:, :512], top[:, 512:]
        cross_t = cross.T
        keep.extend((upper, lower, cross, cross_t))
        keep.append(fnp.copyto(upper, top))
        keep.append(fnp.copyto(lower, cross_t))
        _ipt_product_ab(a1, b1, bottom, 6)
        last = out[512:, 512:]
        keep.extend((last, fnp.copyto(last, bottom)))
        return out
    finally:
        _TR_LAST[0] = None
        keep.clear()


def _gov_sandwich(A, B, inner):
    if 'full' in _GOV_ACTIONS and _gov_fits(A, B, inner, ((1024, 1024),)):
        return _gov_full(A, B)
    if 'two' in _GOV_ACTIONS and _gov_fits(A, B, inner, ((512, 1024), (512, 512))):
        return _gov_two(A, B)
    return None


def _sym_upper(A, B):
    if _GOV['on']:
        out = _gov_sandwich(A, B, 1024)
        if out is not None:
            return out
    return _GOV_PARENT_SYM(A, B)


def _psdm_sym_upper(A, B, m):
    if _GOV['on']:
        out = _gov_sandwich(A, B, m)
        if out is not None:
            _PSDM_STATS['inner'] += 1
            return out
    return _GOV_PARENT_PSDM(A, B, m)


def _gov_predict(self, *args, **kwargs):
    _GOV.update(on=False, r0=_gov_residual(), layer=None, projected=None)
    try:
        return _GOV_PARENT_PREDICT(self, *args, **kwargs)
    finally:
        if _GOV['on']:
            _gov_switch(False)


Estimator.predict = _gov_predict
