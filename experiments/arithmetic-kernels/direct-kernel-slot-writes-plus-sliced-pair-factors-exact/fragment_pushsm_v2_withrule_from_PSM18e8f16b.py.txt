class _SmxKernel(_AstraSm40Kernel):
    def group_views(self, kind, shape):
        # exact-size node scratch at the kernel arena's start (the read's base caps a circuit at 125 nodes; P here has 146)
        key = (self.record['name'], kind, tuple(shape))
        cache = self.M._sm40_group_cache
        if key not in cache:
            groups = self.circuits[kind][1]
            sizes = [len(nodes) for _, nodes in groups]
            total = sum(sizes)
            rows, cols = int(shape[0]), int(shape[1])
            allnodes = fnp.reshape(self.M._SCRATCH[0][:total * rows * cols], (total, rows, cols))
            cuts = []
            at = 0
            for size in sizes[:-1]:
                at += size
                cuts.append(at)
            cache[key] = (tuple(fnp.split(allnodes, cuts, axis=0)), tuple(fnp.unstack(allnodes, axis=0)))
        return cache[key]

    def side(self, kind, L, P, rows, cols):
        key = (self.family[0], kind, L, P, rows, cols)
        cache = self.M._sm40_cache
        if key in cache:
            return cache[key]
        frame = self.M._sm40_frames[self.family[0]]['sides'][kind]
        T, chunks, G, basis, levels = _astra_sm40_slice_side(fnp, frame, P, rows, cols)
        ops = []
        if kind == 0:
            for j in range(1, L + 1):
                X = chunks[j]
                V = G if j == 1 else levels[j]
                ops.extend(((fnp.subtract, X[4], X[0], X[5]), (fnp.subtract, V[2:7:4], X[5], V[3:0:-2])))
        elif kind == 1:
            for j in range(1, L + 1):
                Y = chunks[j]
                V = G if j == 1 else levels[j]
                ops.extend(((fnp.add, Y[1], Y[0], Y[5]), (fnp.subtract, V[5:2:-2], V[2:6:3], V[6:3:-2])))
        else:
            for j in range(L, 1, -1):
                Z = chunks[j]
                V = levels[j]
                ops.extend(((fnp.add, Z[0], V[5:1:-3], V[5:None:-5]), (fnp.add, Z[5], Z[4], Z[5]), (fnp.add, Z[5], V[3:0:-2], V[2:4]),
                            (fnp.add, Z[4], Z[6], Z[1])))
        value = (T, chunks, G, basis, ops)
        cache[key] = value
        return value

    def __call__(self, a, b, L, P, old=None):
        M = self.M
        m, k, n = int(a.shape[0]), int(a.shape[1]), int(b.shape[1])
        ar, ak, an = self.record['shape']
        u = 1 << L
        cm, ck, cn = _rc_math.ceil(m / ar / u) * u, _rc_math.ceil(k / ak / u) * u, _rc_math.ceil(n / an / u) * u
        P = self.family[2]
        assert old is None and tuple(b.shape) == (k, n) and L == self.family[1]
        assert cm // u <= self.family[3] and ck // u <= self.family[4] and cn // u <= self.family[5]
        assert M._sm40_owner is M._SCRATCH[0]
        M._TR_LAST[0] = None
        self.cache.clear()
        self.stats = dict(temp_live=0, temp_peak=0, leaf_bytes=0)
        aa, bb, av, bv, pad_dest = _astra_sm40_padding_views(M, m, k, n, 0, cm, ck, cn, ar, ak, an)
        for zero in pad_dest[4]:
            fnp.copyto(zero, 0)
        fnp.copyto(pad_dest[0], a)
        fnp.copyto(pad_dest[1], b)
        outerkey = (self.family[0], L, cm, ck, cn, P)
        if outerkey not in M._sm40_outer_cache:
            LA = M._sm40_outer['A'][:, :cm, :ck]
            RB = M._sm40_outer['B'][:, :ck, :cn]
            PC = M._sm40_outer['C'][:, :cm, :cn]
            pc = tuple(fnp.unstack(PC, axis=0))
            cuts = list(range(P, 40, P))
            parts = tuple(zip(fnp.split(LA, cuts, axis=0), fnp.split(RB, cuts, axis=0), fnp.split(PC, cuts, axis=0)))
            M._sm40_outer_cache[outerkey] = (LA, RB, PC, pc, parts)
        LA, RB, PC, pc, parts = M._sm40_outer_cache[outerkey]
        _astra_sm40_linear_grouped(fnp, self.circuits['L'], av, (cm, ck), self.stats, out=LA, prepared=self.group_views('L', (cm, ck)))
        _astra_sm40_linear_grouped(fnp, self.circuits['R'], bv, (ck, cn), self.stats, out=RB, prepared=self.group_views('R', (ck, cn)))
        try:
            for index, (A, B, C) in enumerate(parts):
                self.bb_batch(A, B, C, L, (outerkey, index))
        except BaseException:
            M._sm40_cache.clear()
            raise
        finally:
            self.cache.clear()
            M._BH_HOLD.clear()
            M._TR_LAST[0] = None
        packed = _astra_sm40_linear_grouped(fnp, self.circuits['P'], pc, (cm, cn), self.stats, prepared=self.group_views('P', (cm, cn)))
        out = fnp.reshape(fnp.transpose(fnp.reshape(packed, (ar, an, cm, cn)), (0, 2, 1, 3)), (ar * cm, an * cn))
        return fnp.copy(out[:m, :n]) if tuple(out.shape) != (m, n) else out


_SMX_KERNELS = {}


def _smx_register(family):
    family = tuple(family)
    name, L, P, r, k, c = family
    assert 7 ** L * P * (r * k + k * c + r * c) <= 391602176 and P in (1, 2, 4, 5, 8, 10, 20, 40)
    _SMX_FAMILIES[name] = family
    _SMX_ENV._sm40_owner = None


def _smx_bind():
    # Frames: every family from offset 0 of the kernel arena. Outer stacks and padding: ONE flat buffer each, sized to the largest
    # family's total, with each family's views carved from its start (one _smx_call at a time, as for the arena).
    env, arena = _SMX_ENV, _SCRATCH[0]
    env._sm40_owner = None
    env._sm40_cache, env._sm40_io_cache, env._sm40_outer_cache, env._sm40_group_cache = {}, {}, {}, {}
    env._sm40_frames = {name: _astra_sm40_bind_family(fnp, env, arena, f) for name, f in _SMX_FAMILIES.items()}
    dims = {name: (f[3] << f[1], f[4] << f[1], f[5] << f[1]) for name, f in _SMX_FAMILIES.items()}
    outer_n = max(40 * (r * k + k * c + r * c) for r, k, c in dims.values())
    pad_n = max(36 * (r * k + k * c) for r, k, c in dims.values())
    outer, pad = fnp.empty((outer_n,), dtype=fnp.float32), fnp.empty((pad_n,), dtype=fnp.float32)
    env._smx_outer, env._smx_padding = {}, {}
    for name, (r, k, c) in dims.items():
        a, b = 40 * r * k, 40 * k * c
        env._smx_outer[name] = dict(A=fnp.reshape(outer[:a], (40, r, k)), B=fnp.reshape(outer[a:a + b], (40, k, c)),
                                    C=fnp.reshape(outer[a + b:a + b + 40 * r * c], (40, r, c)))
        env._smx_padding[name] = ((fnp.reshape(pad[:36 * r * k], (6 * r, 6 * k)),
                                   fnp.reshape(pad[36 * r * k:36 * r * k + 36 * k * c], (6 * k, 6 * c))), {}, {})
    env._sm40_extra_leaf_bytes = 0
    env._SCRATCH = _SCRATCH
    env._sm40_owner = arena
    _SMX_STATS['binds'] += 1


def _smx_fits(record_name, family_name, m, k, n):
    f = _SMX_FAMILIES.get(family_name)
    if f is None or record_name not in _SMX_RECORDS:
        return False
    ar, ak, an = _SMX_RECORDS[record_name]['shape']
    u = 1 << f[1]
    return -(-m // (ar * u)) <= f[3] and -(-k // (ak * u)) <= f[4] and -(-n // (an * u)) <= f[5]


def _smx_call(record_name, family_name, a, b):
    if _SMX_ENV._sm40_owner is not _SCRATCH[0]:
        _smx_bind()
    kernel = _SMX_KERNELS.get(record_name)
    if kernel is None:
        kernel = _SMX_KERNELS[record_name] = _SmxKernel(_SMX_ENV, _SMX_RECORDS[record_name])
    kernel.family = _SMX_FAMILIES[family_name]
    env = _SMX_ENV
    env._sm40_outer = env._smx_outer[family_name]
    env._sm40_padding, env._sm40_padding_grid, env._sm40_padding_cache = env._smx_padding[family_name]
    env._sm40_outer_bytes = 4 * sum(int(x.shape[0]) * int(x.shape[1]) * int(x.shape[2]) for x in env._sm40_outer.values())
    _SMX_STATS['calls'] += 1
    return kernel(a, b, kernel.family[1], kernel.family[2])

def _pushsm_route(m, k, n):
    if not (_SMX_ON and _PUSHSM_ON) or n <= 1024 or not _smx_fits('signed_executable_0', 'push_L5', m, k, n):
        return False
    if _PUSHSM_FORCE:
        return True
    bits = _PUSHSM_RULE.get((m, k), 0)
    return bool(bits >> (0 if n < 4096 else 1 if n < 6144 else 2 if n < 8192 else 3) & 1)


def _pushsm_mm(a, b):
    if not (_AB_ON and _WINOGRAD and _SCRATCH is not None and len(a.shape) == len(b.shape) == 2
            and str(a.dtype) == str(b.dtype) == 'float32' and _pushsm_route(int(a.shape[0]), int(a.shape[1]), int(b.shape[1]))):
        return _mm(a, b)
    _PUSHSM_STATS['routed'] += 1
    return _smx_call('signed_executable_0', 'push_L5', a, b)
