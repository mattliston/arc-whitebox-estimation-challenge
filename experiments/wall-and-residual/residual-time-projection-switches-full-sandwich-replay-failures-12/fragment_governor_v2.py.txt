# Residual governor. The platform fails a network whose residual (participant Python outside flopscope calls) passes 0.4 s.
# At the start of layers 6..14 this reads the network's residual so far from the open flopscope BudgetContext's live
# residual_wall_time_s (a local clock read: no server request; flopscope-client 0.12.0 = 0.12.1 _budget.py) and projects the end
# residual as residual / _GOV_FRAC[layer], the share of the network's requests made before that layer (S2tr boundary traces,
# nets 35/100; residual instructions run 1:1 with requests). Above _GOV_CAP_S the rest of the network computes each eligible
# covariance sandwich as ONE full tree product (about 315 fewer requests a layer, rounding-level change, flops up) instead of
# the tiled upper blocks. No context, no float, or any exception: inert. A network that never switches runs exactly as S2tr.
_GOV_CAP_S = 0.385
_GOV_FRAC = {6: 0.266, 7: 0.324, 8: 0.394, 9: 0.465, 10: 0.539, 11: 0.643, 12: 0.716, 13: 0.786, 14: 0.880}
_GOV_FORCE = None            # tests only: the layer start that switches whatever the projection
_GOV = dict(on=False, r0=None, layer=None, projected=None)
_GOV_PARENT_SYM = _sym_upper
_GOV_PARENT_PSDM = _psdm_sym_upper
_GOV_PARENT_BEGIN = _w5_port_begin
_GOV_PARENT_PREDICT = Estimator.predict


def _gov_residual():
    try:
        r = flops._budget._active_context.residual_wall_time_s
    except Exception:
        return None
    return r if type(r) is float else None


def _w5_port_begin(scope, network, layer, weight, previous):
    if not _GOV['on']:
        if _GOV_FORCE is not None:
            if layer == _GOV_FORCE:
                _GOV.update(on=True, layer=layer)
        elif _GOV['r0'] is not None and layer in _GOV_FRAC:
            r = _gov_residual()
            if r is not None and r - _GOV['r0'] > _GOV_CAP_S * _GOV_FRAC[layer]:
                _GOV.update(on=True, layer=layer, projected=(r - _GOV['r0']) / _GOV_FRAC[layer])
    return _GOV_PARENT_BEGIN(scope, network, layer, weight, previous)


def _gov_full(A, B, inner):
    # One 1024-square product; G10 mirrored from G01 as in the parent's upper-block semantics.
    if not (_AB_ON and _WINOGRAD and _SCRATCH is not None and inner in (896, 960, 1024)
            and len(A.shape) == len(B.shape) == 2
            and tuple(A.shape) == (1024, inner) and tuple(B.shape) == (inner, 1024)
            and str(A.dtype) == str(B.dtype) == 'float32'
            and _ipt_fits(6, 1024, inner, 1024)):
        return None
    keep = []
    _TR_LAST[0] = None
    try:
        out = fnp.empty((1024, 1024), dtype=fnp.float32)
        _ipt_product_ab(A, B, out, 6)
        upper, lower = out[:512, 512:], out[512:, :512]
        upper_t = upper.T
        keep.extend((upper, lower, upper_t))
        keep.append(fnp.copyto(lower, upper_t))
        return out
    finally:
        _TR_LAST[0] = None
        keep.clear()


def _sym_upper(A, B):
    if _GOV['on']:
        out = _gov_full(A, B, 1024)
        if out is not None:
            return out
    return _GOV_PARENT_SYM(A, B)


def _psdm_sym_upper(A, B, m):
    if _GOV['on']:
        out = _gov_full(A, B, m)
        if out is not None:
            _PSDM_STATS['inner'] += 1
            return out
    return _GOV_PARENT_PSDM(A, B, m)


def _gov_predict(self, *args, **kwargs):
    _GOV.update(on=False, r0=_gov_residual(), layer=None, projected=None)
    try:
        return _GOV_PARENT_PREDICT(self, *args, **kwargs)
    finally:
        _GOV['on'] = False


Estimator.predict = _gov_predict
